export const encodeLook = (s: LookSpec) => `${s.g}.${s.o}.${s.c}.${s.h}.${s.s}`;

/** Parses "g.o.c.h.s"; anything out of range (or a hair style the gender cannot pick) is rejected or fixed */
export function parseLook(raw: unknown): LookSpec | null {
  if (typeof raw !== 'string') return null;
  const p = raw.split('.').map((x) => Number(x));
  if (p.length !== 5 || p.some((n) => !Number.isInteger(n) || n < 0)) return null;
  const [g, o, c, h, s] = p;
  if (g > 1 || o >= OUTFITS.length || c >= 4 || h >= HAIR_STYLES.length || s >= SKIN_TONES.length) return null;
  return { g, o, c, h: HAIR_FOR[g].includes(h) ? h : HAIR_FOR[g][0], s };
}

const hash = (s: string) => {
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) h = Math.imul(h ^ s.charCodeAt(i), 16777619);
  return h >>> 0;
};

/** A stable random look for anyone who has not chosen one (and for the decorative kneelers) */
export function specFor(id: string): LookSpec {
  const h = hash(id);
  const g = h % 2;
  const hairs = HAIR_FOR[g];
  return { g, o: (h >> 3) % OUTFITS.length, c: (h >> 7) % 4, h: hairs[(h >> 11) % hairs.length], s: (h >> 15) % SKIN_TONES.length };
}
