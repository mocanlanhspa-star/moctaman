// Every sound in the hall is synthesised in the browser (original, no copyright issue), plus one music file.
// Music tracks are mixes of layers: drone, bamboo flute, wind chimes, rain, a deep chant pad, a slow wooden fish.
// Crickets and a trickle of water belong to the place, not to a track: they get louder in the garden.
export type TrackId = string;
/** A track is either a mix of generated layers, or one or more recorded files in public/assets/audio */
export type TrackInfo = { id: TrackId; name: string; desc: string; group: string; file?: string; files?: string[] };
const OWN = 'Nhạc do Sếp Mộc làm';

type Layer = 'bgm' | 'drone' | 'flute' | 'chimes' | 'rain' | 'choir';
type MixMap = Partial<Record<Layer | 'mo', number>>;
const FILE_MIX: MixMap = { bgm: 1, drone: 0.1 }; // a recorded track: the file, with a hint of drone underneath
const MIX: Record<string, MixMap> = {
  'tinh-tam': { bgm: 1, drone: 0.25, chimes: 0.3 },
  'sao-truc': { flute: 1, drone: 0.6, chimes: 0.5 },
  'mua-dem': { rain: 1, chimes: 0.8, drone: 0.25 },
  'dem-sen': { drone: 0.35, chimes: 0.3, flute: 0.35 },
  'mo-niem': { choir: 1, mo: 1, drone: 0.4 },
};
const BASE: Record<Layer | 'mo', number> = { bgm: 0.9, drone: 1, flute: 0.55, chimes: 0.5, rain: 0.5, choir: 0.5, mo: 1 };
const FLUTE_NOTES = [293.66, 329.63, 369.99, 440, 493.88, 587.33, 659.25, 739.99]; // D major pentatonic
const CHIME_NOTES = [1174.66, 1318.51, 1567.98, 1760, 2093];

class HallAudio {
  private ctx?: AudioContext;
  private master?: GainNode;
  private musicBus?: GainNode;
  private envBus?: GainNode;
  private reverbIn?: ConvolverNode;
  private layers: Partial<Record<Layer, GainNode>> = {};
  private bgm?: HTMLAudioElement;
  private track: TrackId = 'tinh-tam';
  private zone: 'hall' | 'yard' = 'yard';
  private musicOn = true;
  private volume = 0.8;
  private started = false;

  // Must be called from a click/tap (browsers block sound before that)
  start() {
    if (this.started) return;
    this.started = true;
    const Ctx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    const ctx = (this.ctx = new Ctx());
    const master = (this.master = ctx.createGain());
    master.gain.value = this.volume;
    master.connect(ctx.destination);
    this.musicBus = ctx.createGain();
    this.musicBus.connect(master);
    this.envBus = ctx.createGain();
    this.envBus.connect(master);

    const rev = (this.reverbIn = ctx.createConvolver());
    const len = Math.floor(ctx.sampleRate * 2.8);
    const ir = ctx.createBuffer(2, len, ctx.sampleRate);
    for (let c = 0; c < 2; c++) {
      const d = ir.getChannelData(c);
      for (let i = 0; i < len; i++) d[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / len, 3.2);
    }
    rev.buffer = ir;
    const revOut = ctx.createGain();
    revOut.gain.value = 0.55;
    rev.connect(revOut).connect(master);

    const noise = ctx.createBuffer(1, ctx.sampleRate * 3, ctx.sampleRate);
    const nd = noise.getChannelData(0);
    for (let i = 0; i < nd.length; i++) nd[i] = Math.random() * 2 - 1;
    const noiseSrc = () => {
      const s = ctx.createBufferSource();
      s.buffer = noise;
      s.loop = true;
      s.start();
      return s;
    };
    const layer = (name: Layer) => {
      const g = ctx.createGain();
      g.gain.value = 0;
      g.connect(this.musicBus!);
      this.layers[name] = g;
      return g;
    };
    const filter = (type: BiquadFilterType, f: number, q = 0.7) => {
      const b = ctx.createBiquadFilter();
      b.type = type;
      b.frequency.value = f;
      b.Q.value = q;
      return b;
    };

    // drone: three soft voices under a low-pass
    const drone = layer('drone');
    const lp = filter('lowpass', 420);
    lp.connect(drone);
    [110, 164.8, 220.5].forEach((f, i) => {
      const o = ctx.createOscillator();
      o.type = i === 2 ? 'sine' : 'sawtooth';
      o.frequency.value = f;
      o.detune.value = (i - 1) * 4;
      const g = ctx.createGain();
      g.gain.value = i === 2 ? 0.05 : 0.035;
      o.connect(g).connect(lp);
      o.start();
    });
    // rain
    const rainG = ctx.createGain();
    rainG.gain.value = 0.35;
    noiseSrc().connect(filter('highpass', 900)).connect(filter('lowpass', 7000)).connect(rainG).connect(layer('rain'));
    // choir pad: saw stack through two moving vowel formants
    const choir = layer('choir');
    const f1 = filter('bandpass', 650, 5);
    const f2 = filter('bandpass', 1050, 7);
    const choirIn = ctx.createGain();
    choirIn.gain.value = 0.16;
    [110, 165, 220, 330].forEach((f, i) => {
      const o = ctx.createOscillator();
      o.type = 'sawtooth';
      o.frequency.value = f;
      o.detune.value = (i - 1.5) * 5;
      o.connect(choirIn);
      o.start();
    });
    choirIn.connect(f1).connect(choir);
    choirIn.connect(f2).connect(choir);
    const lfo = ctx.createOscillator();
    lfo.frequency.value = 0.08;
    const lfoG = ctx.createGain();
    lfoG.gain.value = 120;
    lfo.connect(lfoG).connect(f1.frequency);
    lfo.start();
    // music file
    const bgmLayer = layer('bgm');
    this.bgm = new Audio();
    ctx.createMediaElementSource(this.bgm).connect(bgmLayer);
    this.bgm.addEventListener('ended', () => {
      // a playlist moves on to its next file (a single file simply loops)
      if (this.playlist.length > 1) {
        this.playIdx = (this.playIdx + 1) % this.playlist.length;
        this.loadFile();
      }
    });
    layer('flute');
    layer('chimes');

    // the place itself: crickets and trickling water
    [4150, 4620, 5080].forEach((f, i) => {
      const o = ctx.createOscillator();
      o.frequency.value = f;
      const g = ctx.createGain();
      g.gain.value = 0.006;
      const chirp = ctx.createOscillator();
      chirp.type = 'square';
      chirp.frequency.value = 6.4 + i * 1.1;
      const depth = ctx.createGain();
      depth.gain.value = 0.006;
      chirp.connect(depth).connect(g.gain);
      const gate = ctx.createGain();
      gate.gain.value = 0.5;
      const slow = ctx.createOscillator();
      slow.frequency.value = 0.21 + i * 0.07;
      const slowDepth = ctx.createGain();
      slowDepth.gain.value = 0.5;
      slow.connect(slowDepth).connect(gate.gain);
      o.connect(g).connect(gate).connect(this.envBus!);
      o.start();
      chirp.start();
      slow.start();
    });
    const water = ctx.createGain();
    water.gain.value = 0.2;
    noiseSrc().connect(filter('bandpass', 900, 1.2)).connect(water).connect(this.envBus);
    const water2 = ctx.createGain();
    water2.gain.value = 0.07;
    noiseSrc().connect(filter('bandpass', 2300, 3)).connect(water2).connect(this.envBus);
    const ripple = ctx.createOscillator();
    ripple.frequency.value = 0.9;
    const rippleG = ctx.createGain();
    rippleG.gain.value = 0.08;
    ripple.connect(rippleG).connect(water.gain);
    ripple.start();

    this.applyMix();
    this.scheduleFlute();
    this.scheduleChimes();
    this.scheduleFish();
  }

  // ---------------------------------------------------------------- controls
  get currentTrack() {
    return this.track;
  }

  setTrack(id: TrackId) {
    this.track = id;
    this.applyMix();
  }

  setMusic(on: boolean) {
    this.musicOn = on;
    this.applyMix();
  }

  private playlist: string[] = [];
  private playIdx = 0;
  private mix(): MixMap {
    return MIX[this.track] ?? FILE_MIX;
  }
  private loadFile() {
    if (!this.bgm) return;
    this.bgm.src = `/assets/audio/${this.playlist[this.playIdx]}`;
    this.bgm.loop = this.playlist.length === 1;
    if (this.musicOn) this.bgm.play().catch(() => {});
  }

  setVolume(v: number) {
    this.volume = v;
    if (this.master && this.ctx) this.master.gain.setTargetAtTime(v, this.ctx.currentTime, 0.1);
  }

  setZone(z: 'hall' | 'yard') {
    this.zone = z;
    this.applyMix();
  }

  private target(name: Layer | 'mo') {
    return this.musicOn ? (this.mix()[name] ?? 0) * BASE[name] : 0;
  }

  private applyMix() {
    if (!this.ctx) return;
    const t = this.ctx.currentTime;
    (Object.keys(this.layers) as Layer[]).forEach((name) => this.layers[name]!.gain.setTargetAtTime(this.target(name), t, 0.9));
    const env = (this.zone === 'yard' ? 0.5 : 0.08) + (this.track === 'dem-sen' ? 0.4 : 0);
    this.envBus?.gain.setTargetAtTime(env, t, 1.2);
    if (this.bgm) {
      const info = TRACKS.find((x) => x.id === this.track);
      const files = info?.files ?? (info?.file ? [info.file] : []);
      if (files.length && this.musicOn) {
        if (this.playlist.join('|') !== files.join('|')) {
          this.playlist = files;
          this.playIdx = 0;
          this.loadFile();
        } else this.bgm.play().catch(() => {});
      } else {
        if (!files.length) this.playlist = [];
        window.setTimeout(() => this.target('bgm') === 0 && this.bgm?.pause(), 2500);
      }
    }
  }

  // ---------------------------------------------------------------- generative parts
  private note(freq: number, layer: Layer, vol: number, attack: number, hold: number, type: OscillatorType, vibrato = 0) {
    if (!this.ctx || !this.layers[layer]) return;
    const t0 = this.ctx.currentTime;
    const o = this.ctx.createOscillator();
    o.type = type;
    o.frequency.value = freq;
    if (vibrato) {
      const v = this.ctx.createOscillator();
      v.frequency.value = 5.2;
      const vg = this.ctx.createGain();
      vg.gain.value = vibrato;
      v.connect(vg).connect(o.frequency);
      v.start(t0);
      v.stop(t0 + attack + hold + 2.2);
    }
    const g = this.ctx.createGain();
    g.gain.setValueAtTime(0.0001, t0);
    g.gain.exponentialRampToValueAtTime(vol, t0 + attack);
    g.gain.setValueAtTime(vol, t0 + attack + hold);
    g.gain.exponentialRampToValueAtTime(0.0001, t0 + attack + hold + 2);
    o.connect(g);
    g.connect(this.layers[layer]!);
    if (this.reverbIn) g.connect(this.reverbIn);
    o.start(t0);
    o.stop(t0 + attack + hold + 2.2);
  }

  private fluteAt = 3;
  private scheduleFlute() {
    window.setTimeout(() => {
      if (this.target('flute') > 0.03) {
        this.fluteAt = Math.max(0, Math.min(FLUTE_NOTES.length - 1, this.fluteAt + Math.round((Math.random() - 0.5) * 4)));
        const f = FLUTE_NOTES[this.fluteAt];
        this.note(f, 'flute', 0.22, 0.28, 0.5 + Math.random() * 1.4, 'sine', 3.5);
        this.note(f * 2, 'flute', 0.04, 0.35, 0.4, 'sine', 3);
      }
      this.scheduleFlute();
    }, 1600 + Math.random() * 3200);
  }
  private scheduleChimes() {
    window.setTimeout(() => {
      if (this.target('chimes') > 0.03) {
        this.note(CHIME_NOTES[Math.floor(Math.random() * CHIME_NOTES.length)], 'chimes', 0.12, 0.004, 0.05, 'sine');
        if (Math.random() < 0.4) window.setTimeout(() => this.note(CHIME_NOTES[Math.floor(Math.random() * CHIME_NOTES.length)], 'chimes', 0.08, 0.004, 0.05, 'sine'), 220);
      }
      this.scheduleChimes();
    }, 2500 + Math.random() * 5500);
  }
  private scheduleFish() {
    window.setTimeout(() => {
      if (this.target('mo') > 0.03) this.fish(0.16);
      this.scheduleFish();
    }, 1150);
  }

  // ---------------------------------------------------------------- one-shot sounds
  private tone(freq: number, vol: number, decay: number, type: OscillatorType = 'sine', delay = 0) {
    if (!this.ctx || !this.master) return;
    const t0 = this.ctx.currentTime + delay;
    const osc = this.ctx.createOscillator();
    const g = this.ctx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    g.gain.setValueAtTime(0.0001, t0);
    g.gain.exponentialRampToValueAtTime(vol, t0 + 0.012);
    g.gain.exponentialRampToValueAtTime(0.0001, t0 + decay);
    osc.connect(g);
    g.connect(this.master);
    if (this.reverbIn) g.connect(this.reverbIn);
    osc.start(t0);
    osc.stop(t0 + decay + 0.05);
  }

  // Big bronze bell: inharmonic partials with long decay
  bell(volume = 1) {
    const base = 196;
    [1, 2.01, 2.76, 4.07, 5.4, 8.93].forEach((m, i) => this.tone(base * m, (0.34 * volume) / (i + 1.2), 7 / (1 + i * 0.55), 'sine'));
  }

  // Wooden fish: short hollow knock
  fish(volume = 1) {
    this.tone(640, 0.5 * volume, 0.14, 'triangle');
    this.tone(430, 0.35 * volume, 0.2, 'sine');
  }

  // Small chime when incense is planted or a sutra is finished
  chime() {
    [880, 1318.5, 1760].forEach((f, i) => this.tone(f, 0.12, 2.2, 'sine', i * 0.11));
  }
}

export const hallAudio = new HallAudio();
